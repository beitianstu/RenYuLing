    // 获取下一次铃声时间
    private DateTime? GetNextBellTime()
    {
        var s = scheduler.schedule;
        if (s == null || s.Count == 0)
            return null;

        var now = DateTime.Now;

        var next = s
            .Where(c => DateTime.Today.Add(c.Time) >= now)
            .OrderBy(c => c.Time)
            .FirstOrDefault();


        if (next == null)
            return null;

        return DateTime.Today.Add(next.Time);
    }

    private void UpdateBellFileExistState()
    {
        var isStartBellExist = scheduler.isStartBellExist;
        var isEndBellExist = scheduler.isEndBellExist;

        if (!btnChooseStartBell.Enabled)
        {
            txtStartBell.Text = "bell文件夹中存在铃声文件";
            txtStartBell.Enabled = false;
        }

        if (isEndBellExist) btnChooseEndBell.Enabled = false;

        if (!btnChooseEndBell.Enabled)
        {
            txtEndBell.Text = "bell文件夹中存在铃声文件";
            txtEndBell.Enabled = false;
        }
    }

    // 刷新下一次铃声与倒计时
    private void UpdateNextBellUI()
    {
        var now = DateTime.Now;
        var nextBell = GetNextBellTime();
        // Console.WriteLine(nextBell.ToString());

        if (nextBell == null)
        {
            lblNextBell.Text = "今天课程已结束";
            lblCountdown.Text = "";
            return;
        }


        var remain = nextBell.Value - now;

        lblNextBell.Text = $"下一次铃声：{nextBell.Value:HH:mm}";
        lblCountdown.Text = $"倒计时：{remain.Hours:D2}:{remain.Minutes:D2}:{remain.Seconds:D2}";
    }

    // 显示课表
    private void UpdateScheduleDisplay()
    {
        var s = scheduler.schedule;

        if (s == null || s.Count == 0)
        {
            txtboxClassTable.Text = "课表为空";
            return;
        }

        var sb = new StringBuilder();
        sb.AppendLine("今日课表：");
        var t = "default";
        foreach (var item in s)
        {
            t = item.Type switch
            {
                "start" => "上课",
                "end" => "下课",
                _ => t
            };
            sb.AppendLine($"{item.Time:hh\\:mm}     {t}");
        }

        txtboxClassTable.Text = sb.ToString();
    }

    //pictureBox任禹翰敲击动画
    public static void Hit(
        PictureBox pb,
        int durationMs = 220,
        double amplitude = 0.18,
        double damping = 7.0,
        double cycles = 2.2)
    {
        if (pb == null) throw new ArgumentNullException(nameof(pb));
        if (durationMs <= 0) durationMs = 1;

        pb.SizeMode = PictureBoxSizeMode.StretchImage;

        if (Running.TryGetValue(pb, out var oldTimer))
        {
            oldTimer.Stop();
            oldTimer.Dispose();
            Running.Remove(pb);
        }

        var maxHeight = pb.Height;
        var fixedBottom = pb.Bottom;

        var sw = Stopwatch.StartNew();
        var timer = new Timer { Interval = 15 };
        Running[pb] = timer;

        timer.Tick += (_, _) =>
        {
            if (pb.IsDisposed)
            {
                timer.Stop();
                timer.Dispose();
                Running.Remove(pb);
                return;
            }

            var t = sw.Elapsed.TotalMilliseconds / durationMs;
            if (t >= 1) t = 1;

            var u = t;
            var scaleY = 1.0 - amplitude * Math.Exp(-damping * u) * Math.Cos(2.0 * Math.PI * cycles * u);

            if (scaleY > 1.0) scaleY = 1.0;
            if (scaleY < 0.0) scaleY = 0.0;

            var h = (int)Math.Round(maxHeight * scaleY);
            if (h > maxHeight) h = maxHeight;

            pb.Height = h;
            pb.Top = fixedBottom - pb.Height;

            if (t >= 1)
            {
                pb.Height = maxHeight;
                pb.Top = fixedBottom - pb.Height;

                timer.Stop();
                timer.Dispose();
                Running.Remove(pb);
            }
        };

        timer.Start();
    }
